import logging

import requests
from django.conf import settings

logger = logging.getLogger(__name__)


def _headers():
    return {
        "X-Service-Token": settings.CHAT_SERVICE_TOKEN,
        "Content-Type": "application/json",
    }


def chat_enabled() -> bool:
    return bool(settings.CHAT_SERVICE_URL and settings.CHAT_SERVICE_TOKEN)


def sync_chat_group(group):
    if not chat_enabled():
        logger.warning("Chat sync skipped: CHAT_SERVICE_URL or CHAT_SERVICE_TOKEN is empty")
        return False

    url = f"{settings.CHAT_SERVICE_URL}/internal/groups"

    payload = {
        "group_id": group.id,
        "owner_id": group.teacher_id,
        "title": group.group_name,
        "description": None,
        "is_private": False,
    }

    try:
        response = requests.post(
            url,
            json=payload,
            headers=_headers(),
            timeout=5,
        )
        response.raise_for_status()
        return True
    except Exception as exc:
        logger.error("Chat group sync failed: %s", exc)
        return False


def add_chat_members(group_id: int, user_ids: list[int]):
    if not chat_enabled():
        logger.warning("Chat add members skipped: CHAT_SERVICE_URL or CHAT_SERVICE_TOKEN is empty")
        return False

    clean_ids = [int(uid) for uid in user_ids if uid]
    if not clean_ids:
        return True

    url = f"{settings.CHAT_SERVICE_URL}/internal/groups/{group_id}/members"

    try:
        response = requests.post(
            url,
            json={"user_ids": clean_ids},
            headers=_headers(),
            timeout=5,
        )
        response.raise_for_status()
        return True
    except Exception as exc:
        logger.error("Chat add members failed: %s", exc)
        return False


def remove_chat_member(group_id: int, user_id: int):
    if not chat_enabled():
        return False

    url = f"{settings.CHAT_SERVICE_URL}/internal/groups/{group_id}/members/{user_id}"

    try:
        response = requests.delete(
            url,
            headers=_headers(),
            timeout=5,
        )
        response.raise_for_status()
        return True
    except Exception as exc:
        logger.error("Chat remove member failed: %s", exc)
        return False

from django.core.exceptions import ValidationError
from django.db.models import Max
import os


def validate_file(file):
    max_size = 10 * 1024 * 1024  # 10 MB

    allowed_extensions = [".pdf", ".jpg", ".jpeg", ".png", ".doc", ".docx"]

    ext = os.path.splitext(file.name)[1].lower()

    if file.size > max_size:
        raise ValidationError("Файл слишком большой. Максимум 10 MB.")

    if ext not in allowed_extensions:
        raise ValidationError("Недопустимый формат файла.")

def build_leaderboard(students):
    result = []

    for student in students:
        best_results = (
            student.test_results
            .values("test")
            .annotate(best_score=Max("score"))
        )

        tests_total_score = sum(item["best_score"] for item in best_results)

        homeworks_done = student.homework_answers.count()

        rating_points = tests_total_score + homeworks_done * 5

        result.append({
            "student_id": student.id,
            "full_name": student.user.full_name,
            "email": student.user.email,
            "group_id": student.group.id,
            "group_name": student.group.group_name,
            "tests_total_score": tests_total_score,
            "homeworks_done": homeworks_done,
            "rating_points": rating_points,
        })

    result.sort(
        key=lambda x: (-x["rating_points"], -x["tests_total_score"], x["full_name"])
    )

    return result

    def create_superuser(self, username, email=None, password=None, **extra_fields):
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)
        extra_fields.setdefault("is_active", True)
        extra_fields["role"] = "teacher"

        if extra_fields.get("is_staff") is not True:
            raise ValueError("Superuser must have is_staff=True.")
        if extra_fields.get("is_superuser") is not True:
            raise ValueError("Superuser must have is_superuser=True.")

        return super().create_superuser(username, email=email, password=password, **extra_fields)

class PasswordResetConfirmSerializer(serializers.Serializer):
    uid = serializers.CharField()
    token = serializers.CharField()
    new_password = serializers.CharField(write_only=True, min_length=8)
    confirm_password = serializers.CharField(write_only=True, min_length=8)

    def validate(self, attrs):
        if attrs["new_password"] != attrs["confirm_password"]:
            raise serializers.ValidationError({
                "confirm_password": "Пароли не совпадают."
            })

        try:
            uid = force_str(urlsafe_base64_decode(attrs["uid"]))
            user = UserProfile.objects.get(pk=uid)
        except Exception:
            raise serializers.ValidationError({
                "uid": "Неверная ссылка сброса пароля."
            })

        if not default_token_generator.check_token(user, attrs["token"]):
            raise serializers.ValidationError({
                "token": "Токен недействителен или истёк."
            })

        validate_password(attrs["new_password"], user=user)

        attrs["user"] = user
        return attrs

    def save(self):
        user = self.validated_data["user"]
        user.set_password(self.validated_data["new_password"])
        user.save(update_fields=["password"])
        return user

from rest_framework.permissions import BasePermission, SAFE_METHODS


class IsTeacherPermission(BasePermission):
    def has_permission(self, request, view):
        return (
            request.user
            and request.user.is_authenticated
            and (
                request.user.is_superuser
                or request.user.role == 'teacher'
            )
        )


class IsStudentPermission(BasePermission):
    def has_permission(self, request, view):
        return (
            request.user
            and request.user.is_authenticated
            and request.user.role == 'student'
        )


class IsOwnerPermission(BasePermission):
    def has_object_permission(self, request, view, obj):
        if not request.user or not request.user.is_authenticated:
            return False

        if request.user.is_superuser:
            return True

        if request.method in SAFE_METHODS:
            return True

        if hasattr(obj, 'teacher'):
            return obj.teacher == request.user

        if hasattr(obj, 'created_by'):
            return obj.created_by == request.user

        if hasattr(obj, 'user'):
            return obj.user == request.user

        return False

class PasswordResetRequestAPIView(APIView):
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = PasswordResetRequestSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        email = serializer.validated_data["email"]

        user = UserProfile.objects.filter(
            email__iexact=email,
            is_active=True,
        ).first()

        if user:
            uid = urlsafe_base64_encode(force_bytes(user.pk))
            token = default_token_generator.make_token(user)

            frontend_url = getattr(settings, "FRONTEND_URL", "http://localhost:5173")
            reset_url = f"{frontend_url}/reset-password?uid={uid}&token={token}"

            subject = "Сброс пароля"
            message = (
                "Вы запросили сброс пароля.\n\n"
                f"Перейдите по ссылке:\n{reset_url}\n\n"
                "Если вы не запрашивали сброс пароля, просто игнорируйте это письмо."
            )

            send_mail(
                subject=subject,
                message=message,
                from_email=settings.DEFAULT_FROM_EMAIL,
                recipient_list=[user.email],
                fail_silently=False,
            )

        return Response(
            {
                "detail": "Если такой email существует, ссылка для сброса пароля отправлена."
            },
            status=status.HTTP_200_OK,
        )

            },
            status=status.HTTP_201_CREATED,
        )

            answer.comment = request.data.get("comment", answer.comment)
            answer.save()

        return Response(
            HomeworkAnswerSerializer(answer).data,
            status=status.HTTP_201_CREATED if created else status.HTTP_200_OK,
        )

class StudentTestSubmitAPIView(APIView):
    permission_classes = [IsStudentPermission]

    @transaction.atomic
    def post(self, request, result_id):
        serializer = StudentTestSubmitSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        result = get_object_or_404(
            StudentTestResult.objects.select_for_update(),
            id=result_id,
            student__user=request.user,
        )

        if result.status == "submitted":
            return Response(
                {"detail": "Тест уже отправлен."},
                status=status.HTTP_403_FORBIDDEN,
            )

        if result.status == "expired":
            return Response(
                {"detail": "Время теста уже истекло."},
                status=status.HTTP_403_FORBIDDEN,
            )

        test = result.test
        student = result.student

        if student.status != "active":
            return Response(
                {"detail": "Студент не активен в группе."},
                status=status.HTTP_403_FORBIDDEN,
            )

        now = timezone.now()

        if test.time_limit_minutes:
            deadline = result.started_at + timedelta(minutes=test.time_limit_minutes)

            if now > deadline:
                result.status = "expired"
                result.save(update_fields=["status"])

                return Response(
                    {"detail": "Время теста истекло."},
                    status=status.HTTP_403_FORBIDDEN,
                )

        submitted_answers = serializer.validated_data["answers"]

        question_ids = [item["question_id"] for item in submitted_answers]

        if len(question_ids) != len(set(question_ids)):
            return Response(
                {"detail": "Нельзя отправлять один вопрос несколько раз."},
                status=status.HTTP_400_BAD_REQUEST,
            )

        valid_question_ids = set(result.question_order)

        if not set(question_ids).issubset(valid_question_ids):
            return Response(
                {"detail": "Некоторые вопросы не относятся к этой попытке."},
                status=status.HTTP_400_BAD_REQUEST,
            )

        if result.answers.exists():
            return Response(
                {"detail": "Ответы для этой попытки уже сохранены."},
                status=status.HTTP_400_BAD_REQUEST,
            )


        score = 0

        for item in submitted_answers:
            question = get_object_or_404(
                TestQuestion,
                id=item["question_id"],
                test=test,
            )

            allowed_answers = result.answer_order.get(str(question.id), [])

            if item["answer_id"] not in allowed_answers:
                return Response(
                    {"detail": "Ответ не относится к этой попытке."},
                    status=status.HTTP_400_BAD_REQUEST,
                )

            answer = get_object_or_404(
                TestAnswer,
                id=item["answer_id"],
                question=question,
            )

            is_correct = answer.is_correct

            if is_correct:
                score += question.points

            StudentTestAnswer.objects.create(
                result=result,
                question=question,
                selected_answer=answer,
                is_correct=is_correct,
            )

        max_score = test.questions.aggregate(total=Sum("points"))["total"] or 0

        result.score = score
        result.max_score = max_score
        result.status = "submitted"
        result.submitted_at = now
        result.save(update_fields=[
            "score",
            "max_score",
            "status",
            "submitted_at",
        ])

        return Response(
            StudentTestResultSerializer(result).data,
            status=status.HTTP_200_OK,
        )

class StudentTestStartAPIView(APIView):
    permission_classes = [IsStudentPermission]

    @transaction.atomic
    def post(self, request, test_id):
        test = get_object_or_404(CourseTest, id=test_id)

        student = get_object_or_404(
            Student,
            user=request.user,
            group=test.group,
            status="active",
        )

        active_attempt = StudentTestResult.objects.filter(
            test=test,
            student=student,
            status="in_progress",
        ).first()

        if active_attempt:
            if test.time_limit_minutes:
                deadline = active_attempt.started_at + timedelta(minutes=test.time_limit_minutes)
                if timezone.now() > deadline:
                    active_attempt.status = "expired"
                    active_attempt.save(update_fields=["status"])
                    active_attempt = None

        if active_attempt:
            result = active_attempt
        else:
            attempts_count = StudentTestResult.objects.select_for_update().filter(
                test=test,
                student=student,
            ).exclude(status="expired").count()

            if attempts_count >= test.max_attempts:
                return Response(
                    {"detail": "Лимит попыток исчерпан"},
                    status=status.HTTP_403_FORBIDDEN,
                )

            questions = list(test.questions.prefetch_related("answers").all())

            if not questions:
                return Response(
                    {"detail": "В тесте нет вопросов."},
                    status=status.HTTP_400_BAD_REQUEST,
                )

            random.shuffle(questions)

            question_order = [q.id for q in questions]
            answer_order = {}

            for question in questions:
                answers = list(question.answers.all())
                random.shuffle(answers)
                answer_order[str(question.id)] = [a.id for a in answers]

            result = StudentTestResult.objects.create(
                test=test,
                student=student,
                attempt_number=StudentTestResult.objects.filter(
                    test=test,
                    student=student,
                ).count() + 1,
                score=0,
                max_score=0,
                status="in_progress",
                question_order=question_order,
                answer_order=answer_order,
            )

        deadline = None
        if test.time_limit_minutes:
            deadline = result.started_at + timedelta(minutes=test.time_limit_minutes)

        questions_payload = []

        ordered_questions = TestQuestion.objects.filter(
            id__in=result.question_order,
            test=test,
        ).prefetch_related("answers")

        questions_map = {q.id: q for q in ordered_questions}

        for question_id in result.question_order:
            question = questions_map.get(question_id)
            if not question:
                continue

            answer_ids = result.answer_order.get(str(question.id), [])
            answers_map = {a.id: a for a in question.answers.all()}

            answers_payload = []
            for answer_id in answer_ids:
                answer = answers_map.get(answer_id)
                if answer:
                    answers_payload.append({
                        "id": answer.id,
                        "text": answer.text,
                    })

            questions_payload.append({
                "id": question.id,
                "text": question.text,
                "points": question.points,
                "answers": answers_payload,
            })

        data = {
            "result_id": result.id,
            "test_id": test.id,
            "title": test.title,
            "description": test.description,
            "attempt": result.attempt_number,
            "time_limit_minutes": test.time_limit_minutes,
            "deadline": deadline,
            "questions": questions_payload,
        }

        return Response(data, status=status.HTTP_200_OK)
